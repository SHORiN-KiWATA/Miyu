```rust
//! Context meter: provider usage anchor first, local estimate as fallback.
use crate::agent::*;

/// Context occupancy at the end of a finished turn: prompt + completion of
/// its last request. `result.usage` is the whole-turn sum across tool rounds
/// and must not be used here.
pub(in crate::agent) fn context_end_tokens(result: &ChatResult) -> Option<u64> {
    if result.usage_estimated {
        return None;
    }
    let usage = result.last_request_usage.as_ref()?;
    let total = usage.prompt_tokens.saturating_add(usage.completion_tokens);
    (total > 0).then_some(total)
}

impl Agent {
    /// Local o200k estimate over the rendered request (the pre-v3 meter).
    pub fn context_tokens_estimate(&self) -> Result<u64> { /* 原 setup.rs:405-412 的函数体 */ }

    /// Provider-reported occupancy of the newest completed turn. A summary row
    /// at the tail (compaction just ran) or an interrupted/estimated turn
    /// yields None. No provider check on purpose: pools rotate per request,
    /// and a foreign tokenizer's real count is still closer than o200k over
    /// CJK.
    fn context_anchor_tokens(&self) -> Result<Option<u64>> {
        Ok(self.state.load_context_anchor()?.map(|anchor| anchor.tokens))
    }

    pub fn effective_context_tokens(&self) -> Result<u64> {
        let estimate = self.context_tokens_estimate()?;
        match self.context_anchor_tokens()? {
            Some(anchor) => {
                // 量尺日志：testkit 用 `context meter` 抓行算误差
                tracing::info!(estimate, anchor, "context meter");
                Ok(anchor)
            }
            None => Ok(estimate),
        }
    }
}
```

### 2.4 语义说明（写进代码注释）
- 锚点 = 最后一次请求的 `prompt + completion`；下一次请求 ≈ 锚点 + 新用户消息 + 瞬态尾巴，而 `effective_context_tokens()` 是用空输入渲染的，所以直接返回锚点即可，不做"锚点后增量估算"。
- 压缩后最后可见行是摘要行 → 无锚点 → 回退估算，直到下一轮完成。
- `completion` 含推理 token（DeepSeek 等不回放推理），锚点略偏高，偏保守方向，可接受。

## 9. 已做的裁决（用户可否决）与坑
裁决：
1. 回灌默认 5 文件 / 4000 tok / 24000 tok，再受 `window/8` 封顶（小窗口自动缩）。
2. 回灌内容与转录提示放在 checkpoint **同一条 user 消息**里，不伪造工具轮。
3. 转录导出默认开；无 read 工具时提示语不提 read。
4. 锚点不校验供应商是否一致（池轮换时反而更稳）。
5. 分析段只在摘要输出帽 ≥ 6000 tok 时开。
6. "User Requests" 节最新 15 条逐条保留，更早的可并成一行（防跨压缩无限膨胀）。
7. 保留 `MIYU_COMPACT_PROMPT_FILE` / `MIYU_COMPACT_ANALYSIS` 两个环境变量作为实验开关（写进 wiki 的"实验性"小节）。

坑：
- `src/agent/tests/context.rs:846` 靠 "context summarization assistant" 识别摘要请求，提示词首行别改。
- `TurnTokens` 别加字段（22 处字面量）；`Turn` 别加字段（固定列序 SELECT）。
- checkpoint 渲染的两处（`history.rs:127` 与 `:333`）必须走同一 helper，否则 fork 前缀与实况字节分叉、缓存全丢。
- 模型可见文本全英文短句（`check-model-english.sh` 只查 descriptions，但 AGENTS 1.5 是硬要求）；中文只进 `i18n::text` 的 UI 文案。
- 回灌读文件在压缩线程里做同步 IO，单文件 ≤ cap×8 字节、最多 5 个，可接受；别引入异步文件读取改流程。
- 相对路径在 daemon 里 `effective_workdir()` 可能落到进程 cwd，解析不到就跳过，不报错。
- 转录文件不删（与 spill 一致）；不进备份；`session delete` 也不管它——写进注释。
